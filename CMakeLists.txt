cmake_minimum_required(VERSION 3.24)
# Match the UTF-8 console used by Visual Studio/Ninja when parsing localized MSVC dependencies.
if(CMAKE_HOST_WIN32)
    execute_process(COMMAND cmd.exe /d /c "chcp 65001 >nul")
endif()
project(ParamForge VERSION 3.0.0 LANGUAGES C CXX)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set_property(GLOBAL PROPERTY PREDEFINED_TARGETS_FOLDER "CMake")
if(DEFINED PARAMFORGE_MSVC_INCLUDES_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${PARAMFORGE_MSVC_INCLUDES_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
option(PARAMFORGE_BUILD_DESKTOP "Build the desktop application" ON)
include(CTest)
set(PARAMFORGE_QT_ROOT "$ENV{PARAMFORGE_QT_ROOT}" CACHE PATH "Qt SDK directory")
if(WIN32 AND NOT PARAMFORGE_QT_ROOT)
    set(PARAMFORGE_QT_ROOT "C:/Qt/6.12.0/msvc2022_64" CACHE PATH "Qt SDK directory" FORCE)
endif()
if(PARAMFORGE_QT_ROOT)
    list(PREPEND CMAKE_PREFIX_PATH "${PARAMFORGE_QT_ROOT}")
endif()
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Xml)
if(BUILD_TESTING)
    find_package(Qt6 6.12 REQUIRED COMPONENTS Test)
endif()
if(PARAMFORGE_BUILD_DESKTOP)
    find_package(Qt6 6.12 REQUIRED COMPONENTS Widgets OpenGLWidgets Concurrent)
endif()
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(miniz URL https://github.com/richgel999/miniz/archive/refs/tags/3.1.2.tar.gz URL_HASH SHA256=98468f8924934b723276680f85238b6c78bf1f8b49b4459cc9b7214a20e2e9fb DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json miniz)
add_library(ParamForge.Core STATIC
    src/core/model.h src/core/geometry.h
    src/core/model.cpp src/core/tolerances.cpp src/core/engine.cpp src/core/tooling.cpp
    src/core/planning.cpp src/core/postprocessor.cpp src/core/verifier.cpp src/core/operations.cpp
    src/resources/resources.qrc)
target_include_directories(ParamForge.Core PUBLIC src)
target_link_libraries(ParamForge.Core PUBLIC Qt6::Core)
target_compile_definitions(ParamForge.Core PUBLIC QT_NO_KEYWORDS PARAMFORGE_VERSION="${PROJECT_VERSION}")
add_library(ParamForge.Files STATIC src/files/projects.h src/files/projects.cpp src/files/cards.cpp)
target_link_libraries(ParamForge.Files PUBLIC ParamForge.Core Qt6::Xml PRIVATE miniz nlohmann_json::nlohmann_json)
if(WIN32)
    target_link_libraries(ParamForge.Files PRIVATE gdi32)
endif()
if(PARAMFORGE_BUILD_DESKTOP)
    add_library(ParamForge.Desktop STATIC
        src/desktop/window.h src/desktop/part_view.h src/desktop/process_view.h
        src/desktop/inspection_view.h src/desktop/part_3d.h src/desktop/controls.h
        src/desktop/window.cpp src/desktop/part_view.cpp
        src/desktop/process_view.cpp src/desktop/inspection_view.cpp
        src/desktop/part_3d.cpp src/desktop/controls.cpp)
    target_link_libraries(ParamForge.Desktop PUBLIC ParamForge.Files Qt6::Widgets Qt6::OpenGLWidgets Qt6::Concurrent)
    add_executable(ParamForge WIN32 src/desktop/main.cpp)
    if(WIN32)
        enable_language(RC)
        target_sources(ParamForge PRIVATE src/resources/paramforge.rc)
    endif()
    target_link_libraries(ParamForge PRIVATE ParamForge.Desktop)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT ParamForge)
endif()
if(BUILD_TESTING)
    add_executable(ParamForge.Tests tests/core_tests.cpp)
    target_link_libraries(ParamForge.Tests PRIVATE ParamForge.Files Qt6::Test miniz)
    add_test(NAME core COMMAND ParamForge.Tests)
    if(PARAMFORGE_BUILD_DESKTOP)
        add_executable(ParamForge.UiTests tests/ui_tests.cpp)
        target_link_libraries(ParamForge.UiTests PRIVATE ParamForge.Desktop Qt6::Test)
        add_test(NAME ui COMMAND ParamForge.UiTests)
    endif()
endif()
if(WIN32)
    foreach(target ParamForge ParamForge.Tests ParamForge.UiTests)
        if(TARGET ${target})
            add_custom_command(TARGET ${target} POST_BUILD
                COMMAND Qt6::windeployqt
                    "$<IF:$<CONFIG:Debug>,--debug,--release>"
                    --no-compiler-runtime --no-translations --no-quick-import
                    --skip-plugin-types generic,networkinformation,tls
                    "$<TARGET_FILE:${target}>"
                COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/translations"
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "$<TARGET_FILE_DIR:Qt6::qmake>/../translations/qtbase_ru.qm"
                    "$<TARGET_FILE_DIR:${target}>/translations/qtbase_ru.qm"
                VERBATIM)
        endif()
    endforeach()
endif()
if(MSVC)
    foreach(target ParamForge.Core ParamForge.Files ParamForge.Desktop ParamForge ParamForge.Tests ParamForge.UiTests)
        if(TARGET ${target})
        target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive-)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        endif()
    endforeach()
endif()
foreach(target ParamForge.Core ParamForge.Files ParamForge.Desktop ParamForge ParamForge.Tests ParamForge.UiTests)
    if(TARGET ${target})
        get_target_property(project_sources ${target} SOURCES)
        source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${project_sources})
    endif()
endforeach()
set_target_properties(ParamForge.Core ParamForge.Files PROPERTIES FOLDER "Libraries")
if(TARGET ParamForge.Desktop)
    set_property(TARGET ParamForge.Desktop PROPERTY FOLDER "Libraries")
endif()
if(TARGET ParamForge.Tests)
    set_property(TARGET ParamForge.Tests PROPERTY FOLDER "Tests")
endif()
if(TARGET ParamForge.UiTests)
    set_property(TARGET ParamForge.UiTests PROPERTY FOLDER "Tests")
endif()
set_property(TARGET miniz PROPERTY FOLDER "Dependencies")
